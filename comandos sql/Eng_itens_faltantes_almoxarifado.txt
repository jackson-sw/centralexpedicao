-- Relatório de itens faltantes [Almoxarifado]
-- Itens FILHO dos subconjuntos (PAI) da estrutura do projeto, na maior revisão.
-- Usado em backend/routes/relatorios.js (SELECT_ITENS_ESTRUTURA_ALMOXARIFADO — lá a
-- quantidade já é somada por código de item do filho).
-- Colunas do relatório: Código = Produto_Filho, Und = Und_Filho,
-- Estrutura = Qtd_Filho_Total.

SELECT
    PAI.ENG_EstruturaProduto          AS Estrutura,
    PAI.ENG_EstruturaNumero           AS Versao_Pai,
    PAI.ENG_EstruturaItemPosicao      AS Posicao_Pai,
    PAI.ENG_EstruturaItemProduto      AS Produto_Pai,
    PP.PRO_Descricao                  AS Descricao_Pai,
    PAI.ENG_EstruturaItemQuantidade   AS Qtd_Pai,
    FILHO.ENG_EstruturaNumero         AS Versao_Filho,
    FILHO.ENG_EstruturaItemPosicao    AS Posicao_Filho,
    FILHO.ENG_EstruturaItemProduto    AS Produto_Filho,
    PF.PRO_Descricao                  AS Descricao_Filho,
    FILHO.ENG_EstruturaItemTipo       AS Tipo_Filho,
    FILHO.ENG_EstruturaItemDimUnd     AS Und_Filho,
    FILHO.ENG_EstruturaItemDimPecas   AS Pecas_Filho,
    FILHO.ENG_EstruturaItemQuantidade AS Qtd_Filho_Unit,
    FILHO.ENG_EstruturaItemQuantidade * PAI.ENG_EstruturaItemQuantidade AS Qtd_Filho_Total
FROM [dbo].[ENG_ESTRUTURAITEM] PAI
INNER JOIN [dbo].[ENG_ESTRUTURAITEM] FILHO
    ON FILHO.ENG_EstruturaProduto = PAI.ENG_EstruturaItemProduto
   AND FILHO.ENG_EstruturaNumero = (
        SELECT MAX(e2.ENG_EstruturaNumero)
        FROM [dbo].[ENG_ESTRUTURA] e2
        WHERE e2.ENG_EstruturaProduto = FILHO.ENG_EstruturaProduto
   )
INNER JOIN [dbo].[PRO_PRODUTO] PF
    ON PF.PRO_Codigo = FILHO.ENG_EstruturaItemProduto
LEFT JOIN [dbo].[PRO_PRODUTO] PP
    ON PP.PRO_Codigo = PAI.ENG_EstruturaItemProduto
WHERE PAI.ENG_EstruturaItemProduto LIKE '250614-TAV%'
  AND PAI.ENG_EstruturaNumero = (
        SELECT MAX(e2.ENG_EstruturaNumero)
        FROM [dbo].[ENG_ESTRUTURA] e2
        WHERE e2.ENG_EstruturaProduto = PAI.ENG_EstruturaProduto
   )
   AND FILHO.ENG_EstruturaItemLetra NOT IN ('C', 'M', 'S', 'T')
ORDER BY
    PAI.ENG_EstruturaItemPosicao,
    PAI.ENG_EstruturaItemProduto,
    FILHO.ENG_EstruturaItemPosicao;
